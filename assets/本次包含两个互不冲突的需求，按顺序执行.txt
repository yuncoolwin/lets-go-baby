async createDropIn(userId: string, dto: { child_id: string; class_id: string; course_type: string; date: string; teacher_id?: string }) {
const denied = await this.canAccessClass(userId, dto.class_id);
if (denied) return { error: true, code: 403, msg: denied };
if (!dto.child_id || !dto.class_id || !dto.course_type || !dto.date) {
return { error: true, code: 400, msg: '参数不完整' };
}
const { data: dup } = await this.client
.from('drop_in_records')
.select('id')
.eq('child_id', dto.child_id)
.eq('date', dto.date)
.eq('course_type', dto.course_type)
.limit(1);
if (dup && dup.length > 0) {
return { error: true, code: 409, msg: '该幼儿当天已添加过此课程的临时来园' };
}
const { data, error } = await this.client
.from('drop_in_records')
.insert({
child_id: dto.child_id,
class_id: dto.class_id,
course_type: dto.course_type,
date: dto.date,
teacher_id: dto.teacher_id || null,
})
.select()
.single();
if (error) throw error;
return data;
}
async getDropIns(childId: string) {
if (!childId) return [];
const { data: rows } = await this.client
.from('drop_in_records')
.select('id, child_id, class_id, course_type, date, created_at')
.eq('child_id', childId)
.order('date', { ascending: false })
.order('created_at', { ascending: false });
if (!rows || rows.length === 0) return [];
const { data: atts } = await this.client
.from('attendance')
.select('date, course_type, status')
.eq('child_id', childId);
const attMap = new Map<string, string>();
(atts || []).forEach(a => {
if (a.course_type) attMap.set(${a.date}__${a.course_type}, a.status);
});
return rows.map(r => ({
id: r.id,
date: r.date,
course_type: r.course_type,
status: attMap.get(${r.date}__${r.course_type}) || null,
}));
}
